<script>
(function () {
  "use strict";

  /* ============ 曲谱：小星星 ============
     用 MIDI 音高表示：60=C4  62=D4  64=E4  65=F4  67=G4  69=A4
     一共 6 句，每句 7 个音，共 42 个音
  ======================================= */
  var SONG = [
    60, 60, 67, 67, 69, 69, 67,
    65, 65, 64, 64, 62, 62, 60,
    67, 67, 65, 65, 64, 64, 62,
    67, 67, 65, 65, 64, 64, 62,
    60, 60, 67, 67, 69, 69, 67,
    65, 65, 64, 64, 62, 62, 60
  ];

  var TOTAL = SONG.length;

  /* ============ 音频引擎（纯 Web Audio，无需联网） ============ */
  var AudioCtxClass = window.AudioContext || window.webkitAudioContext;
  var ctx = null;

  // 谐波配比，模拟钢琴音色
  var HARMONICS = [1, 0.42, 0.2, 0.11, 0.06, 0.03];

  function initAudio() {
    if (!ctx && AudioCtxClass) {
      ctx = new AudioCtxClass();
    }
    if (ctx && ctx.state === "suspended") {
      ctx.resume();
    }
  }

  function playNote(midi) {
    if (!ctx) return;

    var t = ctx.currentTime;
    var freq = 440 * Math.pow(2, (midi - 69) / 12);

    // 总输出 + 音量包络
    var out = ctx.createGain();
    out.connect(ctx.destination);
    out.gain.setValueAtTime(0.0001, t);
    out.gain.exponentialRampToValueAtTime(0.26, t + 0.012);
    out.gain.exponentialRampToValueAtTime(0.11, t + 0.35);
    out.gain.exponentialRampToValueAtTime(0.0001, t + 2.2);

    for (var i = 0; i < HARMONICS.length; i++) {
      var osc = ctx.createOscillator();
      osc.type = "sine";
      osc.frequency.value = freq * (i + 1);

      var g = ctx.createGain();
      g.gain.value = HARMONICS[i];

      osc.connect(g);
      g.connect(out);

      osc.start(t);
      osc.stop(t + 2.3);
    }
  }

  /* ============ 状态 ============ */
  var index = 0;
  var finished = false;

  /* ============ DOM ============ */
  var pianoEl   = document.getElementById("piano");
  var fillEl    = document.getElementById("fill");
  var countEl   = document.getElementById("count");
  var overlayEl = document.getElementById("overlay");
  var restartEl = document.getElementById("restart");
  var againEl   = document.getElementById("again");

  // 生成琴键（8 个白键，纯装饰，按哪个都一样）
  var KEY_COUNT = 8;
  var keyEls = [];
  for (var i = 0; i < KEY_COUNT; i++) {
    var k = document.createElement("button");
    k.className = "key";
    k.type = "button";
    k.setAttribute("aria-label", "琴键 " + (i + 1));
    pianoEl.appendChild(k);
    keyEls.push(k);
  }

  /* ============ 界面更新 ============ */
  function updateUI() {
    var percent = (index / TOTAL) * 100;
    fillEl.style.width = percent + "%";
    countEl.textContent = "已弹 " + index + " / " + TOTAL + " 个音";
  }

  function flash(el) {
    if (!el) return;
    el.classList.add("pressed");
    setTimeout(function () {
      el.classList.remove("pressed");
    }, 130);
  }

  /* ============ 核心：任意键 → 下一个目标音 ============ */
  function press(el) {
    if (finished) return;

    initAudio();

    playNote(SONG[index]);

    index++;
    updateUI();
    flash(el);

    if (index >= TOTAL) {
      finished = true;
      setTimeout(function () {
        overlayEl.classList.add("show");
      }, 900);
    }
  }

  /* ============ 事件绑定 ============ */
  keyEls.forEach(function (el) {
    el.addEventListener("pointerdown", function (e) {
      e.preventDefault();
      press(el);
    });
  });

  // 电脑键盘也能玩：按任意键即可
  var keyCursor = 0;
  document.addEventListener("keydown", function (e) {
    if (e.repeat) return;
    if (e.key === " " || e.key === "Enter") e.preventDefault();
    if (!overlayEl.classList.contains("show")) {
      var el = keyEls[keyCursor % keyEls.length];
      keyCursor++;
      press(el);
    }
  });

  // 重新开始
  function reset() {
    index = 0;
    finished = false;
    keyCursor = 0;
    overlayEl.classList.remove("show");
    updateUI();
  }

  restartEl.addEventListener("click", reset);
  againEl.addEventListener("click", reset);

  /* ============ 初始化 ============ */
  updateUI();
})();
</script>
